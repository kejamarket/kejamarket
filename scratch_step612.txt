// POST /api/viewings/request — Tenant submits a viewing request
app.post('/api/viewings/request', requireAuth, async (req, res) => {
  try {
    const { propertyId, preferredDate, preferredTime, alternateDate, alternateTime, notes } = req.body;
    if (!propertyId || !preferredDate || !preferredTime) {
      return res.status(400).json({ success: false, message: 'Property, preferred date, and preferred time are required.' });
    }

    const property = await store.getPropertyById(propertyId);
    if (!property) {
      return res.status(404).json({ success: false, message: 'Property not found.' });
    }

    const viewingReq = await store.createViewingRequest({
      tenantId: req.user.id,
      propertyId,
      preferredDate,
      preferredTime,
      alternateDate: alternateDate || null,
      alternateTime: alternateTime || null,
      repNotificationChannel: 'in_app',
      metadata: { notes: notes || '' }
    });

    // Notify landlord if landlord phone exists
    const landlordPhone = property.caretaker_phone || property.caretakerPhone || property.phone;
    if (landlordPhone && typeof sendRealSMS === 'function') {
      try {
        sendRealSMS(
          formatPhone(landlordPhone),
          `[KejaMarket] New viewing request for "${property.title || 'your property'}" on ${preferredDate} at ${preferredTime}. Log in to confirm availability within 60s.`
        );
      } catch (smsErr) {
        console.warn('Viewing request SMS notification error:', smsErr.message);
      }
    }

    res.status(201).json({
      success: true,
      message: 'Viewing request received. Property manager has 60 seconds to respond before our team assists.',
      request: viewingReq,
      escalationSeconds: 60
    });
  } catch (err) {
    console.error('Viewing request error:', err);
    res.status(500).json({ success: false, message: err.message });
  }
});

    if (!vr) {
      return res.status(404).json({ success: false, message: 'Viewing request not found.' });
    }

    const isAdmin = req.user.isAdmin || req.user.role === 'admin' || req.user.id === 'usr-admin-01';
    const isTenant = req.user.id === vr.tenant_id;
    const isLandlord = req.user.id === vr.landlord_id;

    if (!isAdmin && !isTenant && !isLandlord) {

      delete sanitizedVr.backup_contact_phone;

    res.status(500).json({ success: false, message: err.message });

// POST /api/viewings/:id/pay — Tenant initiates KSh 200 viewing commitment fee
app.post('/api/viewings/:id/pay', requireAuth, async (req, res) => {
  try {
    const { phone } = req.body;
    if (!phone) {
      return res.status(400).json({ success: false, message: 'M-Pesa phone number is required.' });
    }

    const vr = await store.getViewingRequest(req.params.id);
    if (!vr) {
      return res.status(404).json({ success: false, message: 'Viewing request not found.' });
    }

    if (vr.tenant_id !== req.user.id && !req.user.isAdmin) {
      return res.status(403).json({ success: false, message: 'You can only pay for your own viewing booking.' });
    }

    const feeAmount = 200; // Spec v1.1 KSh 200 commitment fee
    const formattedPhone = formatPhone(phone);

    // If live Daraja configured, dispatch real STK push
    if (hasDarajaCredentials()) {
      try {
        const token = await getMpesaToken();
        const timestamp = getDarajaTimestamp();
        const password = generatePassword(timestamp);

        const payload = {
          BusinessShortCode: PAYBILL,
          Password: password,
          Timestamp: timestamp,
          TransactionType: 'CustomerPayBillOnline',
          Amount: feeAmount,
          PartyA: formattedPhone,
          PartyB: PAYBILL,
          PhoneNumber: formattedPhone,
          CallBackURL: CALLBACK_URL,
          AccountReference: `VIEW-${vr.id.slice(-6).toUpperCase()}`,
          TransactionDesc: `Viewing Fee: ${vr.property_title || 'KejaMarket'}`.slice(0, 32),
        };

        const stkRes = await fetch(`${MPESA_BASE}/mpesa/stkpush/v1/processrequest`, {
          method: 'POST',
          headers: {
            Authorization: `Bearer ${token}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(payload),
        });

        const stkData = await stkRes.json();
        if (stkData.ResponseCode === '0') {
          await store.createTransaction({
            checkoutRequestId: stkData.CheckoutRequestID,
            merchantRequestId: stkData.MerchantRequestID,
            phone: formattedPhone,
            amount: feeAmount,
            itemType: 'viewing_fee',
            itemName: `Viewing Fee: ${vr.property_title}`,
            targetPropertyId: vr.property_id,
            userId: req.user.id,
            status: 'PENDING'
          });

          await store.updateViewingRequest(vr.id, {
            status: 'payment_initiated'
          });

          return res.json({
            success: true,
            checkoutRequestId: stkData.CheckoutRequestID,
            message: `M-Pesa STK push dispatched to ${formattedPhone}. Enter your PIN on your phone.`
          });
        }
      } catch (stkErr) {
        console.warn('Viewing fee STK push error:', stkErr.message);
      }
    }

    // Sandbox / Simulation fallback
    const mockCheckoutId = 'ws_VIEW_' + Date.now() + '_' + Math.random().toString(36).substring(2, 6);
    await store.createTransaction({
      checkoutRequestId: mockCheckoutId,
      merchantRequestId: 'MR_VIEW_' + Date.now(),
      phone: formattedPhone,
      amount: feeAmount,
      itemType: 'viewing_fee',
      itemName: `Viewing Fee: ${vr.property_title}`,
      targetPropertyId: vr.property_id,
      userId: req.user.id,
      status: 'PENDING'
    });

    res.json({
      success: true,
      checkoutRequestId: mockCheckoutId,
      isSimulation: true,
      message: `Prompt sent to ${formattedPhone}. Please verify payment with your M-Pesa receipt.`
    });
  } catch (err) {
    console.error('Viewing payment error:', err);
    res.status(500).json({ success: false, message: err.message });
  }
});

      }
    }
  } catch (escalationErr) {
    console.warn('Escalation worker check:', escalationErr.message);
  }
}, 15000); // Check every 15 seconds
